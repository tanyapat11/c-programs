#include <iostream>
#include <string>

using namespace std;

class Product {
private:
    int productId;
    string productName;
    double price;
    int monthlySales[12];

public:
    void inputDetails() {
        cout << "Enter Product ID: ";
        cin >> productId;
        cin.ignore();
        cout << "Enter Product Name: ";
        getline(cin, productName);
        cout << "Enter Product Price: ";
        cin >> price;
        
        cout << "Enter sales quantity for 12 months:\n";
        for (int i = 0; i < 12; i++) {
            cin >> monthlySales[i];
        }
    }

    int calculateTotalQuantity() const {
        int total = 0;
        for (int i = 0; i < 12; i++) {
            total += monthlySales[i];
        }
        return total;
    }

    void displayDetails() const {
        int totalQty = calculateTotalQuantity();
        double totalBill = totalQty * price;

        cout << "\nProduct ID: " << productId << endl;
        cout << "Product Name: " << productName << endl;
        cout << "Price per Unit: $" << price << endl;
        cout << "Total Quantity Sold: " << totalQty << endl;
        cout << "Total Bill Amount: $" << totalBill << endl;
        cout << "-----------------------------------" << endl;
    }
};

int main() {
    int n;
    cout << "Enter the number of products (up to 50): ";
    cin >> n;

    Product products[50];

    for (int i = 0; i < n; i++) {
        cout << "\nEnter Details for Product " << i + 1 << ":\n";
        products[i].inputDetails();
    }

    cout << "\n========= PRODUCT SALES REPORT =========";
    for (int i = 0; i < n; i++) {
        products[i].displayDetails();
    }

    return 0;
}
